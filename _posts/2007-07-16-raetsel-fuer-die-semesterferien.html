class Base implements IBase{

class Derived extends Base {

public void print(String s) {
super.print(s);
super.print(this.s);
}
}
interface IBase{
String s = “+”;
void print(long n) ;
void print(String s);
void printlt();
}

interface IDerived{
String s = “*”;
void print(long n) ;
void print(String s);
void printlt();
}
